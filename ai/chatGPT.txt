• schedule(static) distributes contiguous ranges of source indices among threads.

void reorder_parallel
(
    const char* src
,
    char* dst
,
    const Dimensions& dims
) {
    if (src == nullptr || dst == nullptr
) {
        throw std::invalid_argument("Null buffer"
);
    }

    const Layout layout = make_layout(dims
);
    const Layout out_layout
 =
        make_layout(reverse_dimensions(dims
));

    const std::size_t rank = dims
.size();
    const std::size_t total = layout
.element_count;

    // Tune this value experimentally on the target machine.
    constexpr std::size_t CHUNK_SIZE = 1U << 16
;
    const std::size_t num_chunks
 =
        total / CHUNK_SIZE + (total % CHUNK_SIZE != 0
);

    #pragma
 omp parallel for schedule(static)
    for (long long chunk_id = 0
;
         chunk_id < static_cast<long long>(num_chunks
);
         ++chunk_id
) {

        const std::size_t begin
 =
            static_cast<std::size_t>(chunk_id) * CHUNK_SIZE
;
        const std::size_t end
 =
            std::
min(begin + CHUNK_SIZE, total
);

        // Recover coordinates only once at the beginning
        // of each chunk.
        Dimensions indices(rank
);
        std::
size_t dst_offset = 0
;

        for (std::size_t d = 0; d < rank; ++d
) {
            indices[d
] =
                (
begin / layout.c_strides[d]) % dims[d
];

            dst_offset +=
                indices[d] * out_layout.c_strides[rank - 1 - d
];
        }

        // Process contiguous source elements.
        for (std::size_t linear = begin
;
             linear < end
;
             ++linear
) {

            dst[dst_offset] = src[linear
];

            if (linear + 1 == end
) {
                break
;
            }

            // Advance the mixed-radix coordinates.
            // The last dimension changes fastest in C order.
            for (std::size_t d = rank; d-- > 0
;) {
                const std::size_t out_stride
 =
                    out_layout.c_strides[rank - 1 - d
];

                ++indices[d
];
                dst_offset += out_stride
;

                if (indices[d] < dims[d
]) {
                    break
;
                }

                // This dimension overflowed: reset and carry.
                dst_offset -= dims[d] * out_stride
;
                indices[d] = 0
;
            }
        }
    }
}
